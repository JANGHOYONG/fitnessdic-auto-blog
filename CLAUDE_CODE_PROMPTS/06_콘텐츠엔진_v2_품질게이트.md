# 06 · 콘텐츠 엔진 v2 — 품질 게이트 적용

## 🎯 목표

기존 `scripts/content-generator.js`를 전면 리라이트한다. 양산형 자동 발행에서 **품질 기준선을 통과해야만 발행되는** 엔진으로 바꾼다. 이 엔진이 있는 한, AI 생성 글도 애드센스·네이버가 인식할 수 있는 "사람이 감수한 고품질 콘텐츠"가 된다.

---

## 🔑 품질 기준선 7대 원칙 (이 중 하나라도 미달 시 `QUALITY_REJECTED` 상태로 큐에 저장)

1. **본문 2,500자 이상 (공백 제외 2,200자 이상)**
2. **1차 출처 최소 3건** — 논문/공식 가이드/정부 자료 URL 포함
3. **저자 Author 모델 연결** — 카테고리에 맞는 전문가 자동 매칭
4. **실사용 경험 요소 1개 이상** — 비교표, 단계별 가이드, 체크리스트, 사례 중 하나
5. **이미지 최소 3장** — 대표 + 본문 최소 2장 (Pexels + 인포그래픽 1장 슬롯)
6. **FAQ 4~6개** — JSON-LD용으로 정확한 Q/A 형식
7. **내부 링크 5개 이상** — 같은 카테고리 기존 글을 자동 검색해 삽입

### 4) `scripts/quality-gate.js`

```javascript
const MIN_CHARS = 2200;
const MIN_SOURCES = 3;
const MIN_IMAGES = 3;
const MIN_FAQS = 4;
const MIN_INTERNAL_LINKS = 5;

function plainTextLength(content) {
  return content.filter(b => b.html).map(b => b.html.replace(/<[^>]+>/g, "")).join("").length;
}

function runQualityGate(post) {
  const reasons = [];
  const charLen = plainTextLength(post.content);
  if (charLen < MIN_CHARS) reasons.push(`본문 짧음 (${charLen}자, 기준 ${MIN_CHARS})`);
  if ((post.sources?.length ?? 0) < MIN_SOURCES) reasons.push(`출처 부족 (${post.sources?.length ?? 0}건, 기준 ${MIN_SOURCES})`);
  if ((post.images?.length ?? 0) < MIN_IMAGES) reasons.push(`이미지 부족`);
  const faqBlock = post.content.find(b => b.type === "faq");
  if (!faqBlock || (faqBlock.items?.length ?? 0) < MIN_FAQS) reasons.push(`FAQ 부족`);
  if (!post.authorId) reasons.push(`저자 미지정`);
  if ((post.internalLinks?.length ?? 0) < MIN_INTERNAL_LINKS) reasons.push(`내부 링크 부족`);
  const hasExperience = post.content.some(b => ["compare","checklist","case"].includes(b.type));
  if (!hasExperience) reasons.push(`체험 요소 없음 (비교표/체크리스트/사례)`);
  // 금지어
  const forbidden = ["치료한다","낫게","기적의","무조건","확실히 효과"];
  const body = post.content.map(b => JSON.stringify(b)).join("");
  const hits = forbidden.filter(w => body.includes(w));
  if (hits.length) reasons.push(`금지어 포함: ${hits.join(", ")}`);
  const score = Math.max(0, 100 - reasons.length * 14);
  return { pass: reasons.length === 0, score, reasons };
}
module.exports = { runQualityGate };
```

### 5) `scripts/source-fetcher.js`

OpenAI에 "이 주제의 1차 출처 후보 5건을 JSON으로"라고 별도 호출하여, 모델이 본문 출처를 "지어내지 않도록" 분리한다. 이후 코드에서 URL이 실제 접속 가능한지 HEAD 체크로 검증 (접속 불가 출처는 제외).

```javascript
async function fetchSourceCandidates(topic) {
  const res = await openai.chat.completions.create({
    model: "gpt-4o-mini",
    messages: [
      { role: "system", content: "주제에 대한 1차 출처(논문, 정부가이드, 학회 공식자료) 5건을 JSON 배열로 반환하세요. 임의 URL 생성 금지. 확실하지 않으면 제목만 반환." },
      { role: "user", content: `주제: ${topic}\n\n출력 예: [{"title":"...","url":"...","publisher":"..."}]` }
    ],
    response_format: { type: "json_object" },
  });
  const raw = JSON.parse(res.choices[0].message.content);
  const candidates = raw.sources ?? raw.items ?? [];
  const verified = [];
  for (const s of candidates) {
    if (!s.url) continue;
    try {
      const r = await fetch(s.url, { method: "HEAD", redirect: "follow" });
      if (r.ok) verified.push({ ...s, accessedAt: new Date().toISOString() });
    } catch {}
  }
  return verified;
}
```

### 8) `scripts/content-generator.js` — 메인 파이프라인

```javascript
const { runQualityGate } = require("./quality-gate");
const { fetchSourceCandidates } = require("./source-fetcher");
const { insertInternalLinks } = require("./internal-linker");
const { factCheck } = require("./fact-checker");

async function generateOne(topic) {
  // 1) 카테고리 결정
  const category = detectCategory(topic);                       // 10개 중 하나
  const { SYSTEM_PROMPT } = require(`./prompts/${category}`);

  // 2) 출처 선수집
  const sources = await fetchSourceCandidates(topic);

  // 3) 저자 자동 매칭 (카테고리별)
  const author = await prisma.author.findFirst({ where: { expertise: { has: category } } });

  // 4) 본문 생성
  const res = await openai.chat.completions.create({
    model: "gpt-4o-mini",
    messages: [
      { role: "system", content: SYSTEM_PROMPT },
      { role: "user", content: `주제: ${topic}\n\n아래 출처를 각 주장에 맞춰 인용하되, 없는 URL을 지어내지 마세요.\n${JSON.stringify(sources)}` }
    ],
    response_format: { type: "json_object" },
    max_tokens: 5000,
  });
  const draft = JSON.parse(res.choices[0].message.content);

  // 5) 이미지 수집 (Pexels)
  const images = await fetchPexelsImages(draft.coverImageQuery, 3);

  // 6) 내부 링크 삽입
  const internalLinks = await insertInternalLinks(draft, category);

  // 7) 팩트체크
  const factReport = await factCheck(draft);

  const post = {
    ...draft,
    authorId: author?.id,
    sources,
    images,
    internalLinks,
    factReport,
    status: "DRAFT",
  };

  // 8) 품질 게이트
  const gate = runQualityGate(post);
  post.qualityScore = gate.score;
  if (!gate.pass) {
    post.status = "QUALITY_REJECTED";
    post.rejectReasons = gate.reasons;
  } else {
    post.status = "REVIEW";      // 사람 감수 큐로
  }

  // 9) DB 저장
  return await prisma.post.create({ data: toPrismaData(post) });
}
```
